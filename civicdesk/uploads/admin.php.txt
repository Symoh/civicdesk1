
<?php
declare(strict_types=1);

$isHttps = !empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off';

session_set_cookie_params([
    'lifetime' => 0,
    'path' => '/',
    'secure' => $isHttps,
    'httponly' => true,
    'samesite' => 'Lax',
]);

session_start();

header('Cache-Control: no-store, no-cache, must-revalidate, max-age=0');
header('Pragma: no-cache');

require_once __DIR__ . '/config.php';

function escape(string $value): string
{
    return htmlspecialchars($value, ENT_QUOTES, 'UTF-8');
}

if (!isset($_SESSION['csrf_token'])) {
    $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
}

if (
    !isset($_SESSION['user_id'], $_SESSION['role']) ||
    $_SESSION['role'] !== 'admin'
) {
    header('Location: login.php');
    exit;
}

$adminId = (int) $_SESSION['user_id'];
$allowedStatuses = ['Pending', 'In Progress', 'Resolved'];
$error = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $token = (string) ($_POST['csrf_token'] ?? '');
    $requestId = filter_var(
        $_POST['request_id'] ?? null,
        FILTER_VALIDATE_INT
    );
    $newStatus = (string) ($_POST['new_status'] ?? '');
    $notes = trim((string) ($_POST['admin_notes'] ?? ''));

    if (!hash_equals($_SESSION['csrf_token'], $token)) {
        $error = 'Your form session expired. Refresh and try again.';
    } elseif ($requestId === false || $requestId < 1) {
        $error = 'Invalid service request.';
    } elseif (!in_array($newStatus, $allowedStatuses, true)) {
        $error = 'Please select a valid status.';
    } elseif (mb_strlen($notes) > 5000) {
        $error = 'Administrator notes must not exceed 5,000 characters.';
    } else {
        try {
            $pdo->beginTransaction();

            // Lock the request to avoid conflicting simultaneous updates.
            $findStmt = $pdo->prepare(
                'SELECT status FROM service_requests
                 WHERE id = :id FOR UPDATE'
            );
            $findStmt->execute(['id' => $requestId]);
            $request = $findStmt->fetch();

            if (!$request) {
                throw new RuntimeException('Service request not found.');
            }

            $oldStatus = (string) $request['status'];

            $updateStmt = $pdo->prepare(
                'UPDATE service_requests
                 SET status = :status,
                     admin_notes = :notes,
                     updated_at = CURRENT_TIMESTAMP
                 WHERE id = :id'
            );

            $updateStmt->execute([
                'status' => $newStatus,
                'notes' => $notes !== '' ? $notes : null,
                'id' => $requestId,
            ]);

            // Record a history entry only when the status actually changes.
            if ($oldStatus !== $newStatus) {
                $historyStmt = $pdo->prepare(
                    'INSERT INTO request_status_history
                        (request_id, changed_by, old_status, new_status, notes)
                     VALUES
                        (:request_id, :changed_by, :old_status, :new_status, :notes)'
                );

                $historyStmt->execute([
                    'request_id' => $requestId,
                    'changed_by' => $adminId,
                    'old_status' => $oldStatus,
                    'new_status' => $newStatus,
                    'notes' => $notes !== '' ? $notes : null,
                ]);
            }

            $pdo->commit();

            $_SESSION['admin_notice'] = $oldStatus === $newStatus
                ? 'Request notes saved. The status was unchanged.'
                : 'Request status updated and history recorded.';

            $_SESSION['csrf_token'] = bin2hex(random_bytes(32));

            header('Location: admin.php');
            exit;

        } catch (Throwable $e) {
            if ($pdo->inTransaction()) {
                $pdo->rollBack();
            }

            error_log('CivicDesk admin update failed: ' . $e->getMessage());
            $error = $e instanceof RuntimeException
                && $e->getMessage() === 'Service request not found.'
                ? 'That service request could not be found.'
                : 'The update failed. Check the server logs and try again.';
        }
    }
}

$notice = $_SESSION['admin_notice'] ?? null;
unset($_SESSION['admin_notice']);

try {
    $statsStmt = $pdo->query(
        "SELECT COUNT(*) AS total,
                COALESCE(SUM(status = 'Pending'), 0) AS pending,
                COALESCE(SUM(status = 'In Progress'), 0) AS in_progress,
                COALESCE(SUM(status = 'Resolved'), 0) AS resolved
         FROM service_requests"
    );
    $stats = $statsStmt->fetch();

    $requestsStmt = $pdo->query(
        'SELECT sr.id, sr.reference_number, sr.category, sr.location,
                sr.description, sr.image_path, sr.status, sr.admin_notes,
                sr.created_at, u.full_name, u.email
         FROM service_requests sr
         INNER JOIN users u ON u.id = sr.citizen_id
         ORDER BY
            CASE sr.status
                WHEN "Pending" THEN 1
                WHEN "In Progress" THEN 2
                ELSE 3
            END,
            sr.created_at DESC
         LIMIT 100'
    );
    $requests = $requestsStmt->fetchAll();

} catch (PDOException $e) {
    error_log('CivicDesk admin dashboard failed: ' . $e->getMessage());
    http_response_code(500);
    exit('Unable to load the administration dashboard.');
}
