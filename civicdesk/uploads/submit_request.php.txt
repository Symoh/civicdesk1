<?php
declare(strict_types=1);

$isHttps = !empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off';

session_set_cookie_params([
    'lifetime' => 0,
    'path' => '/',
    'secure' => $isHttps,
    'httponly' => true,
    'samesite' => 'Lax',
]);

session_start();

header('Cache-Control: no-store, no-cache, must-revalidate, max-age=0');
header('Pragma: no-cache');

require_once __DIR__ . '/config.php';

// Require an authenticated user before submitting a request.
if (!isset($_SESSION['user_id'], $_SESSION['role'])) {
    header('Location: login.php');
    exit;
}

// Administrators should use the admin dashboard.
if ($_SESSION['role'] === 'admin') {
    header('Location: admin.php');
    exit;
}

// Only citizens may submit service requests.
if ($_SESSION['role'] !== 'citizen') {
    http_response_code(403);
    exit('Access denied. Your account does not have permission to submit requests.');
}

function escape(string $value): string
{
    return htmlspecialchars($value, ENT_QUOTES, 'UTF-8');
}

if (!isset($_SESSION['csrf_token'])) {
    $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
}

$categories = [
    'Roads and Transport',
    'Water and Sanitation',
    'Waste Management',
    'Street Lighting',
    'Public Health',
    'Drainage and Flooding',
    'Other',
];

$errors = [];
$category = '';
$location = '';
$description = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $submittedToken = (string) ($_POST['csrf_token'] ?? '');
    $category = trim((string) ($_POST['category'] ?? ''));
    $location = trim((string) ($_POST['location'] ?? ''));
    $description = trim((string) ($_POST['description'] ?? ''));

    if (!hash_equals($_SESSION['csrf_token'], $submittedToken)) {
        $errors[] = 'Your form session expired. Refresh the page and try again.';
    }

    if (!in_array($category, $categories, true)) {
        $errors[] = 'Please select a valid service category.';
    }

    if ($location === '' || mb_strlen($location) > 255) {
        $errors[] = 'Enter the location of the issue (maximum 255 characters).';
    }

    if ($description === '' || mb_strlen($description) < 10
        || mb_strlen($description) > 5000) {
        $errors[] = 'Describe the issue in at least 10 characters and no more than 5,000.';
    }

    // Validate an optional image.
  	$uploadedFile = $_FILES['evidence'] ?? null;
	$temporaryPath = null;
	$savedImagePath = null;

if ($uploadedFile !== null && !is_array($uploadedFile)) {
    $errors[] = 'The uploaded image data is invalid.';
}

    if ($uploadedFile !== null && $uploadedFile['error'] !== UPLOAD_ERR_NO_FILE) {
        if ($uploadedFile['error'] !== UPLOAD_ERR_OK) {
            $errors[] = 'The image upload failed. Please try again.';
        } elseif ($uploadedFile['size'] > 5 * 1024 * 1024) {
            $errors[] = 'The image must be 5 MB or smaller.';
        } elseif (!is_uploaded_file($uploadedFile['tmp_name'])) {
            $errors[] = 'The uploaded file could not be verified.';
        } else {
            $temporaryPath = $uploadedFile['tmp_name'];

            $finfo = new finfo(FILEINFO_MIME_TYPE);
            $mimeType = $finfo->file($temporaryPath);

            $allowedTypes = [
                'image/jpeg' => 'jpg',
                'image/png'  => 'png',
                'image/webp' => 'webp',
            ];

            if (!isset($allowedTypes[$mimeType])) {
                $errors[] = 'Only JPG, PNG and WebP images are allowed.';
            }
        }
    }

    if ($errors === []) {
        $uploadDirectory = __DIR__ . '/uploads/requests';

        if (!is_dir($uploadDirectory)
            && !mkdir($uploadDirectory, 0750, true)
            && !is_dir($uploadDirectory)) {
            $errors[] = 'The upload folder is unavailable. Contact the administrator.';
        }
    }

    if ($errors === []) {
        try {
            // Generate a difficult-to-guess reference number.
            $referenceNumber = 'CD-' . date('Ymd') . '-'
                . strtoupper(bin2hex(random_bytes(4)));

            if ($temporaryPath !== null) {
                $extension = $allowedTypes[$mimeType];
                $filename = bin2hex(random_bytes(16)) . '.' . $extension;
                $destination = $uploadDirectory . '/' . $filename;

                if (!move_uploaded_file($temporaryPath, $destination)) {
                    throw new RuntimeException('Unable to save uploaded evidence.');
                }

                $savedImagePath = 'uploads/requests/' . $filename;
            }

            $stmt = $pdo->prepare(
                "INSERT INTO service_requests
                    (reference_number, citizen_id, category, location,
                     description, image_path)
                 VALUES
                    (:reference_number, :citizen_id, :category, :location,
                     :description, :image_path)"
            );

            $stmt->execute([
                'reference_number' => $referenceNumber,
                'citizen_id' => (int) $_SESSION['user_id'],
                'category' => $category,
                'location' => $location,
                'description' => $description,
                'image_path' => $savedImagePath,
            ]);

            // Rotate the form token and show a one-time success message.
            $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
            $_SESSION['request_success'] = $referenceNumber;

            header('Location: submit_request.php');
            exit;

        } catch (Throwable $e) {
            // Remove a file if its database record could not be saved.
            if ($savedImagePath !== null) {
                $fullPath = __DIR__ . '/' . $savedImagePath;
                if (is_file($fullPath)) {
                    unlink($fullPath);
                }
            }

            error_log('CivicDesk request submission failed: ' . $e->getMessage());
            $errors[] = 'Your request could not be saved. Please try again.';
        }
    }
}

$successReference = $_SESSION['request_success'] ?? null;
unset($_SESSION['request_success']);

            <?php if (is_string($successReference) && $successReference !== ''): ?>
                <div class="message-success" role="status">
                    <h2>Request submitted successfully</h2>
                    <p>Your reference number is:</p>
                    <p><strong><?= escape($successReference) ?></strong></p>
                    <p>Keep this number for your records.</p>
                    <a class="btn" href="dashboard.php">Return to dashboard</a>
                </div>
            <?php endif; ?>

            <?php if ($errors !== []): ?>
                <div class="message-error" role="alert">

                <div class="form-field">
                    <label for="category">Service category</label>
                    <select id="category" name="category" required>
                        <option value="">Choose a category</option>
                        <?php foreach ($categories as $item): ?>
                            <option value="<?= escape($item) ?>"
