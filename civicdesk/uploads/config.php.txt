
<?php
declare(strict_types=1);

/**
 * CivicDesk database configuration.
 *
 * Local development defaults are intended for XAMPP only.
 * Production credentials should be supplied through environment variables.
 */

$dbHost = getenv('CIVICDESK_DB_HOST') ?: '127.0.0.1';
$dbName = getenv('CIVICDESK_DB_NAME') ?: 'civicdesk_db';
$dbUser = getenv('CIVICDESK_DB_USER') ?: 'root';
$dbPass = getenv('CIVICDESK_DB_PASS');

if ($dbPass === false) {
    $dbPass = '';
}

$dsn = "mysql:host={$dbHost};dbname={$dbName};charset=utf8mb4";

try {
    $pdo = new PDO($dsn, $dbUser, $dbPass, [
        PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
        PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
        PDO::ATTR_EMULATE_PREPARES   => false,
    ]);
} catch (PDOException $e) {
    error_log(
        'CivicDesk database connection failed: ' . $e->getMessage()
    );

    if (!headers_sent()) {
        http_response_code(500);
        header('Content-Type: text/plain; charset=UTF-8');
    }

    exit('CivicDesk is temporarily unavailable. Please try again later.');
}
