function escape(string $value): string
{
    return htmlspecialchars($value, ENT_QUOTES, 'UTF-8');
}

function formatDate(?string $value): string
{
    if (!$value) {
        return 'Not available';
    }

    $timestamp = strtotime($value);

    return $timestamp !== false
        ? date('d M Y, h:i A', $timestamp)
        : 'Not available';
}

function statusClass(string $status): string
{
    return match (strtolower(trim($status))) {
        'pending' => 'pending',
        'in progress' => 'in-progress',
        'resolved' => 'resolved',
        default => 'other',
    };
}

$userId = (int) $_SESSION['user_id'];
$fullName = (string) ($_SESSION['full_name'] ?? 'Citizen');

try {
    // Preserve the existing citizen-specific statistics query.
      // Count requests while handling inconsistent status capitalization/spaces.
    $countStmt = $pdo->prepare(
        "SELECT
            COUNT(*) AS total,
            COALESCE(SUM(LOWER(TRIM(status)) = 'pending'), 0) AS pending,
            COALESCE(SUM(LOWER(TRIM(status)) = 'in progress'), 0) AS in_progress,
            COALESCE(SUM(LOWER(TRIM(status)) = 'resolved'), 0) AS resolved
         FROM service_requests
         WHERE citizen_id = :citizen_id"
    );
    $countStmt->execute(['citizen_id' => $userId]);

    $stats = $countStmt->fetch(PDO::FETCH_ASSOC) ?: [
        'total' => 0,
        'pending' => 0,
        'in_progress' => 0,
        'resolved' => 0,
    ];

    // Retrieve only the signed-in citizen's requests.
    // No new database columns or tables are required.
    $requestStmt = $pdo->prepare(
        "SELECT reference_number, category, location, status,
                created_at, updated_at
         FROM service_requests
         WHERE citizen_id = :citizen_id
         ORDER BY created_at DESC
         LIMIT 20"
    );

    $requestStmt->execute(['citizen_id' => $userId]);
    $requests = $requestStmt->fetchAll(PDO::FETCH_ASSOC);

} catch (PDOException $e) {
    error_log('CivicDesk dashboard query failed: ' . $e->getMessage());
    http_response_code(500);
    exit(
        'Unable to load your dashboard at the moment. Please try again later.'
    );
}

<script>
(function () {
    'use strict';

    // Mobile sidebar.
    const sidebar = document.getElementById('cdSidebar');
    const overlay = document.getElementById('cdOverlay');
    const menuButton = document.getElementById('cdMenuButton');

    function setMenu(open) {
        sidebar.classList.toggle('is-open', open);
        overlay.classList.toggle('is-visible', open);
        document.body.classList.toggle('menu-open', open);
        menuButton.setAttribute('aria-expanded', String(open));
        menuButton.setAttribute(
            'aria-label',
            open ? 'Close navigation menu' : 'Open navigation menu'
        );
        menuButton.textContent = open ? '×' : '☰';
    }

    menuButton.addEventListener('click', function () {
        setMenu(!sidebar.classList.contains('is-open'));
    });

    overlay.addEventListener('click', function () {
        setMenu(false);
    });

    sidebar.querySelectorAll('a').forEach(function (link) {
        link.addEventListener('click', function () {
            setMenu(false);
        });
    });

    document.addEventListener('keydown', function (event) {
        if (event.key === 'Escape') {
            setMenu(false);
        }
    });

    window.addEventListener('resize', function () {
        if (window.innerWidth > 760) {
            setMenu(false);
        }
    });

    // Client-side search and status filtering.
    // This filters only the requests already retrieved by PHP.
    const searchInput = document.getElementById('cdSearch');
    const statusFilter = document.getElementById('cdStatusFilter');
    const requestList = document.getElementById('cdRequestList');
    const resultsCount = document.getElementById('cdResultsCount');
    const noMatches = document.getElementById('cdNoMatches');
    const clearButton = document.getElementById('cdClearFilters');

    if (!searchInput || !statusFilter || !requestList) {
        return;
    }

    const cards = Array.from(
        requestList.querySelectorAll('.cd-request-card')
    );

    function filterRequests() {
        const term = searchInput.value.trim().toLocaleLowerCase();
        const selectedStatus = statusFilter.value;
        let visible = 0;

        cards.forEach(function (card) {
            const searchable = (
                card.dataset.search || ''
            ).toLocaleLowerCase();

            const cardStatus = (
                card.dataset.status || ''
            ).toLocaleLowerCase();

            const matchesSearch = searchable.includes(term);
            const matchesStatus = (
                selectedStatus === 'all' ||
                cardStatus === selectedStatus
            );

            const show = matchesSearch && matchesStatus;

            card.classList.toggle('cd-hidden', !show);

            if (show) {
                visible++;
            }
        });

        resultsCount.textContent =
            'Showing ' + visible + ' of ' + cards.length +
            ' requests currently loaded';

        noMatches.classList.toggle('cd-hidden', visible !== 0);
    }

    searchInput.addEventListener('input', filterRequests);
    statusFilter.addEventListener('change', filterRequests);

    if (clearButton) {
        clearButton.addEventListener('click', function () {
            searchInput.value = '';
            statusFilter.value = 'all';
            filterRequests();
            searchInput.focus();
        });
    }
})();
</script>
