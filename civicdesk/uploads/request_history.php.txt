
<?php
declare(strict_types=1);

$isHttps = !empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off';

session_set_cookie_params([
    'lifetime' => 0,
    'path' => '/',
    'secure' => $isHttps,
    'httponly' => true,
    'samesite' => 'Lax',
]);

session_start();

$role = (string) $_SESSION['role'];
$userId = (int) $_SESSION['user_id'];

if (!in_array($role, ['citizen', 'admin'], true)) {
    http_response_code(403);
    exit('Access denied.');
}

function escape(string $value): string
{
    return htmlspecialchars($value, ENT_QUOTES, 'UTF-8');
}

$requestId = filter_var($_GET['id'] ?? null, FILTER_VALIDATE_INT);

if ($requestId === false || $requestId === 0 || $requestId < 1) {
    http_response_code(400);
    exit('A valid request ID is required. Open history from a request in your dashboard.');
}

try {
    // Citizens can only open requests that belong to their own account.
    $requestSql = '
        SELECT sr.id, sr.reference_number, sr.category, sr.location,
               sr.status, sr.created_at, sr.description
        FROM service_requests sr
        WHERE sr.id = :id
    ';

    if ($role === 'citizen') {
        $requestSql .= ' AND sr.citizen_id = :citizen_id';
    }

    $requestStmt = $pdo->prepare($requestSql);
    $requestParams = ['id' => $requestId];

    if ($role === 'citizen') {
        $requestParams['citizen_id'] = $userId;
    }

    $requestStmt->execute($requestParams);
    $request = $requestStmt->fetch();

    if (!$request) {
        http_response_code(404);
        exit('Request not found, or you do not have permission to view it.');
    }

    $historyStmt = $pdo->prepare(
        'SELECT h.old_status, h.new_status, h.notes, h.changed_at,
                u.full_name AS changed_by
         FROM request_status_history h
         LEFT JOIN users u ON u.id = h.changed_by
         WHERE h.request_id = :request_id
         ORDER BY h.changed_at DESC, h.id DESC'
    );

    $historyStmt->execute(['request_id' => $requestId]);
    $history = $historyStmt->fetchAll();

} catch (PDOException $e) {
    error_log('CivicDesk request history failed: ' . $e->getMessage());
    http_response_code(500);
    exit('Unable to load request history. Please try again later.');
}

$backLink = $role === 'admin' ? 'admin.php' : 'dashboard.php';
?>

        }

            overflow-x: auto;
        }
        .history-table {
            width: 100%;
            min-width: 650px;
            border-collapse: collapse;
        }
        .history-table th,
        .history-table td {
            padding: 14px;
            text-align: left;
            vertical-align: top;
            border-bottom: 1px solid #e1e7ef;
        }
        .history-table th {
            background: #f6f8fb;
        }
        .history-empty {
            padding: 20px;
            background: #f6f8fb;
            border-radius: 8px;
        }
    </style>
</head>
<body>

    <div class="container navbar">
        <a class="brand" href="index.php">
            <span class="brand-mark" aria-hidden="true">C</span>
            <span>CivicDesk</span>
        </a>
        <nav class="nav-links" aria-label="Main navigation">
            <a href="<?= escape($backLink) ?>">Back to dashboard</a>
            <a href="logout.php">Sign out</a>
        </nav>
    </div>
</header>

<main class="section">
    <div class="container">
        <h1>Request status history</h1>

        <section class="history-card">
            <h2><?= escape((string) $request['reference_number']) ?></h2>
            <p><strong>Category:</strong> <?= escape((string) $request['category']) ?></p>
            <p><strong>Location:</strong> <?= escape((string) $request['location']) ?></p>
            <p><strong>Current status:</strong> <?= escape((string) $request['status']) ?></p>
            <p><strong>Submitted:</strong>
                <?= escape(date('d M Y H:i', strtotime((string) $request['created_at']))) ?>
            </p>
            <p><strong>Description:</strong><br>
                <?= nl2br(escape((string) $request['description'])) ?>
            </p>
        </section>

        <h2>Recorded changes</h2>

        <?php if ($history === []): ?>
            <div class="history-empty">
                <p>No status changes have been recorded for this request yet.</p>
                <p>The initial submission may not have a history entry; this page displays changes recorded after submission.</p>
            </div>
        <?php else: ?>
            <div class="history-table-wrap">
                <table class="history-table">
                    <thead>
                    <tr>
                        <th>Date and time</th>
                        <th>Previous status</th>
                        <th>New status</th>
                        <th>Changed by</th>
                        <th>Notes</th>
                    </tr>
                    </thead>
                    <tbody>
                    <?php foreach ($history as $entry): ?>
                        <tr>
                            <td>
                                <?= escape(date(
                                    'd M Y H:i',
                                    strtotime((string) $entry['changed_at'])
                                )) ?>
                            </td>
                            <td><?= escape((string) $entry['old_status']) ?></td>
                            <td><?= escape((string) $entry['new_status']) ?></td>
                            <td><?= escape((string) ($entry['changed_by'] ?? 'Unknown')) ?></td>
                            <td><?= nl2br(escape((string) ($entry['notes'] ?? ''))) ?></td>
                        </tr>
                    <?php endforeach; ?>
                    </tbody>
                </table>
            </div>
        <?php endif; ?>

        <p style="margin-top:24px">
            <a class="btn btn-secondary" href="<?= escape($backLink) ?>">Back</a>
        </p>
    </div>
</main>
