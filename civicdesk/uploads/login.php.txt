
<?php
declare(strict_types=1);

// Configure session cookies before starting the session.
$isHttps = !empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off';

session_set_cookie_params([
    'lifetime' => 0,
    'path' => '/',
    'secure' => $isHttps,
    'httponly' => true,
    'samesite' => 'Lax',
]);

session_start();
require_once __DIR__ . '/config.php';

function escape(string $value): string
{
    return htmlspecialchars($value, ENT_QUOTES, 'UTF-8');
}

// Prevent browsers from retaining a cached copy of this page.
header('Cache-Control: no-store, no-cache, must-revalidate, max-age=0');
header('Pragma: no-cache');

if (
    isset($_SESSION['user_id'], $_SESSION['role']) &&
    in_array($_SESSION['role'], ['admin', 'citizen'], true)
) {
    header(
        'Location: ' .
        ($_SESSION['role'] === 'admin' ? 'admin.php' : 'dashboard.php')
    );
    exit;
}

if (!isset($_SESSION['csrf_token'])) {
    $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
}

$error = '';
$email = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $email = strtolower(trim((string) ($_POST['email'] ?? '')));
    $password = (string) ($_POST['password'] ?? '');
    $submittedToken = (string) ($_POST['csrf_token'] ?? '');

    if (!hash_equals($_SESSION['csrf_token'], $submittedToken)) {
        $error = 'Your form session has expired. Please refresh and try again.';
    } elseif (
        !filter_var($email, FILTER_VALIDATE_EMAIL) ||
        strlen($email) > 190 ||
        $password === ''
    ) {
        $error = 'Enter a valid email address and password.';
    } else {
        try {
            $stmt = $pdo->prepare(
                'SELECT id, full_name, email, password_hash, role
                 FROM users
                 WHERE email = :email
                 LIMIT 1'
            );

            $stmt->execute(['email' => $email]);
            $user = $stmt->fetch();

            $validPassword = $user
                && is_string($user['password_hash'])
                && password_verify($password, $user['password_hash']);

            if (
                $validPassword &&
                in_array($user['role'], ['admin', 'citizen'], true)
            ) {
                // Prevent session fixation.
                session_regenerate_id(true);

                $_SESSION['user_id'] = (int) $user['id'];
                $_SESSION['full_name'] = (string) $user['full_name'];
                $_SESSION['user_email'] = (string) $user['email'];
                $_SESSION['role'] = (string) $user['role'];

                // Rotate CSRF token after authentication.
                $_SESSION['csrf_token'] = bin2hex(random_bytes(32));

                $destination = $user['role'] === 'admin'
                    ? 'admin.php'
                    : 'dashboard.php';

                header('Location: ' . $destination);
                exit;
            }

            $error = 'Invalid email address or password.';
        } catch (PDOException $e) {
            error_log('CivicDesk login failed: ' . $e->getMessage());
            $error = 'Sign-in is temporarily unavailable. Please try again.';
        }
    }
}

                </div>
